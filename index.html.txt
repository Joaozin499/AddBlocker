        (function() {
            // Conteúdo do perfil .mobileconfig para AdGuard DNS (Bloqueio de anúncios)
            const profileContent = `<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
    <key>PayloadContent</key>
    <array>
        <dict>
            <key>Name</key>
            <string>AdGuard DNS (Ad-Blocking)</string>
            <key>PayloadDescription</key>
            <string>Configures device to use AdGuard DNS over HTTPS (DoH)</string>
            <key>PayloadDisplayName</key>
            <string>AdGuard DNS (Ad-Blocking)</string>
            <key>PayloadIdentifier</key>
            <string>com.adguard.dns.profile</string>
            <key>PayloadType</key>
            <string>com.apple.dnsSettings.managed</string>
            <key>PayloadUUID</key>
            <string>$(uuidgen)</string>
            <key>PayloadVersion</key>
            <integer>1</integer>
            <key>DNSSettings</key>
            <dict>
                <key>DNSProtocol</key>
                <string>HTTPS</string>
                <key>ServerAddresses</key>
                <array>
                    <string>94.140.14.14</string>
                    <string>94.140.15.15</string>
                </array>
                <key>ServerURL</key>
                <string>https://dns.adguard-dns.com/dns-query</string>
            </dict>
        </dict>
    </array>
    <key>PayloadDescription</key>
    <string>Adiciona o AdGuard DNS (bloqueio de anúncios) ao dispositivo.</string>
    <key>PayloadDisplayName</key>
    <string>AdGuard DNS (Ad-Blocking)</string>
    <key>PayloadIdentifier</key>
    <string>com.yourname.adguard-dns</string>
    <key>PayloadRemovalDisallowed</key>
    <false/>
    <key>PayloadType</key>
    <string>Configuration</string>
    <key>PayloadUUID</key>
    <string>$(uuidgen)</string>
    <key>PayloadVersion</key>
    <integer>1</integer>
</dict>
</plist>`.replace(/\$\(uuidgen\)/g, () => {
                // Gera um UUID v4 válido para cada placeholder
                return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
                    var r = Math.random() * 16 | 0, v = c == 'x' ? r : (r & 0x3 | 0x8);
                    return v.toString(16);
                });
            });

            function downloadProfile() {
                // Cria o blob com o tipo MIME correto que o iOS reconhece
                const blob = new Blob([profileContent], { type: 'application/x-apple-aspen-config' });
                
                // Cria uma URL para o blob
                const url = URL.createObjectURL(blob);
                
                // Cria um link temporário e clica nele
                const a = document.createElement('a');
                a.href = url;
                a.download = 'adguard-dns.mobileconfig';
                document.body.appendChild(a);
                a.click();
                
                // Limpeza
                setTimeout(() => {
                    document.body.removeChild(a);
                    URL.revokeObjectURL(url);
                }, 100);
            }

            // Adiciona o evento de clique ao botão
            document.getElementById('downloadBtn').addEventListener('click', downloadProfile);
        })();
